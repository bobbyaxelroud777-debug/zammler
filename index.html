// Гарантированное определение роли по URL (?dept=РО_ПП7 или ?dept=СП_ПП7) function initRole() { const params = new URLSearchParams(window.location.search); const deptParam = params.get('dept'); const roleBadge = document.getElementById('role-badge'); const adminPanel = document.getElementById('admin-panel'); if (deptParam && deptParam.trim() !== '') { currentRole = 'user'; currentDept = decodeURIComponent(deptParam).replace(/\s+/g, '_'); const formattedName = currentDept.replace('_', ' '); // Прячем панель админа для сотрудников if (adminPanel) adminPanel.classList.add('hidden'); roleBadge.innerHTML = ` Должность: ${formattedName} `; document.getElementById('view-title').innerText = `Задачи для: ${formattedName}`; } else { currentRole = 'admin'; currentDept = null; if (adminPanel) adminPanel.classList.remove('hidden'); roleBadge.innerHTML = ` Режим: Администратор `; document.getElementById('view-title').innerText = `Все задачи сети (11 складов)`; } }
